2-Minute Google Sheets Setup Guide

Follow these 3 quick steps to turn any Google Sheet into your private cloud database (100% free):

  1. Create a new spreadsheet at sheets.new and name it Rent Records.
  2. In the top menu, click Extensions → Apps Script.
  3. Delete any code there and paste the snippet below, then click Deploy → New deployment → Select type: Web app → Who has access: Anyone.
  4. Copy the resulting Web App URL and paste it into RentPulse Settings!
Google Apps Script Code (Code.gs) 
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: "online",
    service: "RentPulse Webhook",
    time: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({status: "error", message: "No data"})).setMimeType(ContentService.MimeType.JSON);
    }
    var data = JSON.parse(e.postData.contents);
    if (data.action === "ping" || data.action === "test_connection") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        code: "PONG",
        service: "RentPulse Webhook",
        time: new Date().toISOString()
      })).setMimeType(ContentService.MimeType.JSON);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Rent_Bills") || ss.insertSheet("Rent_Bills");
    var headers = ["Timestamp", "Bill Number", "Month", "Tenant Name", "Previous Reading (kWh)", "Current Reading (kWh)", "Units Consumed", "Electricity Bill (₹)", "Room Rent (₹)", "Total Amount (₹)", "Paid Amount (₹)", "Balance Due (₹)", "Status", "Payment Mode", "Invoice PDF Link", "Notes"];

    var folderName = "RentPulse_Invoices";
    var folderIter = DriveApp.getFoldersByName(folderName);
    var folder = folderIter.hasNext() ? folderIter.next() : DriveApp.createFolder(folderName);

    if (data.action === "sync_bills" && Array.isArray(data.bills)) {
      sheet.clearContents();
      sheet.appendRow(headers);
      var headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setBackground("#4f46e5").setFontColor("#ffffff").setFontWeight("bold");
      sheet.setFrozenRows(1);

      data.bills.forEach(function(b) {
        var pdfUrl = "";
        try {
          var billNum = b.billNumber || "Bill";
          var fileName = "Invoice_" + billNum + "_" + (b.billingMonth || "Month") + "_" + (b.status || "Status") + ".pdf";
          var existing = folder.getFilesByName(fileName);
          if (existing.hasNext()) {
            pdfUrl = existing.next().getUrl();
          } else {
            var units = (typeof b.unitsConsumed === "number") ? b.unitsConsumed : ((Number(b.currReading) || 0) - (Number(b.prevReading) || 0));
            var html = '
' + (data.owner ? data.owner.name : 'RentPulse') + '
Invoice #' + billNum + ' (' + b.billingMonth + ')
Tenant: ' + (b.tenantName || '') + '
Total: ₹' + (Number(b.totalAmount) || 0).toFixed(2) + ' | Status: ' + b.status + '
';
            var blob = Utilities.newBlob(html, "text/html", fileName).getAs("application/pdf");
            var f = folder.createFile(blob.setName(fileName));
            f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
            pdfUrl = f.getUrl();
          }
        } catch(err) { pdfUrl = "Error: " + err.toString(); }

        sheet.appendRow([
          b.createdAt || new Date().toISOString(), b.billNumber, b.billingMonth, b.tenantName,
          Number(b.prevReading) || 0, Number(b.currReading) || 0, Number(b.unitsConsumed) || 0,
          Number(b.electricityCost) || 0, Number(b.roomRent) || 0, Number(b.totalAmount) || 0,
          Number(b.paidAmount) || 0, Number(b.balanceDue) || 0, b.status, b.paymentMode || "",
          pdfUrl, b.notes || ""
        ]);
      });
      return ContentService.createTextOutput(JSON.stringify({status: "success", count: data.bills.length, folder: folder.getUrl()})).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({status: "ignored"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
